-- ============================================================
-- 4) VIEW profiles_public (display_name)
-- IMPORTANTE: NO usar CREATE OR REPLACE porque p.* cambia al agregar columnas
-- y rompe con ERROR 42P16. La creamos solo si NO existe.
-- ============================================================

do $$
begin
  if not exists (
    select 1
    from pg_class c
    join pg_namespace n on n.oid = c.relnamespace
    where n.nspname = 'public'
      and c.relname = 'profiles_public'
      and c.relkind = 'v'
  ) then
    execute $v$
      create view public.profiles_public as
      select
        p.*,
        case
          when coalesce(trim(p.first_name), '') <> '' then
            trim(coalesce(p.first_name,'') || ' ' || coalesce(p.last_name,''))
          when coalesce(trim(p.full_name), '') <> '' then
            trim(p.full_name)
          else coalesce(p.email, 'Usuario')
        end as display_name
      from public.profiles p;
    $v$;
  end if;
end $$;


-- ============================================================
-- 5) AUTH -> PROFILES bootstrap (handle_new_user)
-- ============================================================
create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  au record;
  v_full_name text;
begin
  select * into au
  from public.allowed_users
  where lower(email) = lower(new.email)
  limit 1;

  if au is null or au.is_active is false then
    return new;
  end if;

  v_full_name :=
    coalesce(
      nullif(trim(coalesce(au.full_name,'')), ''),
      nullif(trim(coalesce(au.first_name,'')), ''),
      new.raw_user_meta_data->>'full_name',
      new.email
    );

  insert into public.profiles (
    id, email,
    full_name, first_name, last_name,
    dni, birth_date, team, job_title, start_date,
    blood_type, emergency_contact_name, emergency_contact_phone,
    role, annual_vacation_days, active
  )
  values (
    new.id, lower(new.email),
    v_full_name, au.first_name, au.last_name,
    au.dni, au.birth_date, au.team, au.job_title, au.start_date,
    au.blood_type, au.emergency_contact_name, au.emergency_contact_phone,
    coalesce(au.role,'user'), coalesce(au.annual_vacation_days,10), true
  )
  on conflict (id) do update
  set
    email = excluded.email,
    full_name = excluded.full_name,
    first_name = excluded.first_name,
    last_name = excluded.last_name,
    dni = excluded.dni,
    birth_date = excluded.birth_date,
    team = excluded.team,
    job_title = excluded.job_title,
    start_date = coalesce(excluded.start_date, public.profiles.start_date),
    blood_type = excluded.blood_type,
    emergency_contact_name = excluded.emergency_contact_name,
    emergency_contact_phone = excluded.emergency_contact_phone,
    role = excluded.role,
    annual_vacation_days = excluded.annual_vacation_days,
    active = true;

  return new;
end;
$$;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
after insert on auth.users
for each row execute function public.handle_new_user();

-- ============================================================
-- 6) RPC ensure_my_profile()
-- ============================================================
create or replace function public.ensure_my_profile()
returns public.profiles
language plpgsql
security definer
set search_path = public
as $$
declare
  v_email text := lower(auth.jwt() ->> 'email');
  a record;
  p public.profiles;
  v_full_name text;
begin
  if v_email is null or v_email = '' then
    raise exception 'Missing email in JWT';
  end if;

  select * into a
  from public.allowed_users
  where lower(email) = v_email
  limit 1;

  if a is null then raise exception 'NOT_ALLOWED'; end if;
  if a.is_active is false then raise exception 'NOT_ACTIVE'; end if;

  v_full_name := coalesce(
    nullif(trim(coalesce(a.full_name,'')), ''),
    nullif(trim(coalesce(a.first_name,'')), ''),
    v_email
  );

  insert into public.profiles (
    id, email,
    full_name, first_name, last_name,
    dni, birth_date, team, job_title, start_date,
    blood_type, emergency_contact_name, emergency_contact_phone,
    role, annual_vacation_days, active
  )
  values (
    auth.uid(), v_email,
    v_full_name, a.first_name, a.last_name,
    a.dni, a.birth_date, a.team, a.job_title, a.start_date,
    a.blood_type, a.emergency_contact_name, a.emergency_contact_phone,
    a.role, coalesce(a.annual_vacation_days,10), true
  )
  on conflict (id) do update
  set
    email = excluded.email,
    full_name = coalesce(excluded.full_name, public.profiles.full_name),
    first_name = coalesce(excluded.first_name, public.profiles.first_name),
    last_name = coalesce(excluded.last_name, public.profiles.last_name),
    dni = coalesce(excluded.dni, public.profiles.dni),
    birth_date = coalesce(excluded.birth_date, public.profiles.birth_date),
    team = excluded.team,
    job_title = excluded.job_title,
    start_date = coalesce(excluded.start_date, public.profiles.start_date),
    blood_type = coalesce(excluded.blood_type, public.profiles.blood_type),
    emergency_contact_name = coalesce(excluded.emergency_contact_name, public.profiles.emergency_contact_name),
    emergency_contact_phone = coalesce(excluded.emergency_contact_phone, public.profiles.emergency_contact_phone),
    role = excluded.role,
    annual_vacation_days = excluded.annual_vacation_days,
    active = true
  returning * into p;

  return p;
end;
$$;

revoke all on function public.ensure_my_profile() from public;
grant execute on function public.ensure_my_profile() to authenticated;

-- ------------------------------------------------------------
-- DAY HELPERS
-- ------------------------------------------------------------

create or replace function public.business_days_between(d1 date, d2 date)
returns integer
language sql
immutable
as $$
  select count(*)::int
  from generate_series(least(d1, d2), greatest(d1, d2), interval '1 day') as g(day)
  where extract(isodow from g.day) between 1 and 5;
$$;

create or replace function public.calendar_days_between(d1 date, d2 date)
returns integer
language sql
immutable
as $$
  select (greatest(d1, d2) - least(d1, d2) + 1)::int;
$$;

create or replace function public.vacation_days_between(d1 date, d2 date)
returns integer
language plpgsql
stable
as $$
declare
  v_mode text;
begin
  select vacation_count_mode into v_mode
  from public.org_settings
  where id = 1;

  if v_mode = 'calendar_days' then
    return public.calendar_days_between(d1, d2);
  end if;

  return public.business_days_between(d1, d2);
end;
$$;

create or replace function public.vacation_days_overlap(
  abs_from date,
  abs_to date,
  range_from date,
  range_to date
)
returns int
language plpgsql
stable
as $$
declare
  a_from date := greatest(abs_from, range_from);
  a_to   date := least(abs_to, range_to);
begin
  if a_from > a_to then
    return 0;
  end if;

  return public.vacation_days_between(a_from, a_to);
end;
$$;


-- ------------------------------------------------------------
-- ENTITLEMENT
-- ------------------------------------------------------------

create or replace function public.vacation_entitlement_days_at_date(
  p_user_id uuid,
  p_at date
)
returns int
language plpgsql
stable
as $$
declare
  v_start date;
  v_years int;
  v_schema jsonb;
  v_days int := 14;
begin
  select start_date into v_start
  from public.profiles
  where id = p_user_id;

  if v_start is null then
    return null;
  end if;

  v_years := date_part('year', age(p_at, v_start))::int;
  if v_years < 0 then v_years := 0; end if;

  select entitlement_schema into v_schema
  from public.org_settings
  where id = 1;

  select (e->>'days')::int
  into v_days
  from jsonb_array_elements(v_schema) e
  where (e->>'min_years')::int <= v_years
  order by (e->>'min_years')::int desc
  limit 1;

  return coalesce(v_days, 14);
end;
$$;

-- ============================================================
-- BALANCE ACUMULATIVO (SIN VENCIMIENTO) + MIGRACIÓN (saldo inicial)
-- ============================================================
create or replace function public.get_my_vacation_balance_at(p_at date)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  v_uid uuid := auth.uid();

  v_start_date date;

  -- MIGRACIÓN
  v_migration_date date;
  v_initial_available numeric := 0;
  v_has_migration boolean := false;

  -- RANGO BASE para contar consumos
  v_from_date date;

  -- acumulación/grants (legacy total desde start_date)
  annual_days int;

  earned_at_p numeric := 0;
  earned_before_migration numeric := 0;

  -- output "granted" efectivo
  granted_total numeric := 0;

  -- consumos
  used_total numeric := 0;
  reserved_total numeric := 0;
  available_total numeric := 0;

  -- helpers para cálculo earned (reusa tu lógica actual)
  v_cut date;
  months_worked int;
  y int;
  ann date;
  start_year int;
  end_year int;

  -- para la segunda evaluación
  v_eval date;
begin
  if v_uid is null then
    raise exception 'Not authenticated';
  end if;

  perform public.ensure_my_profile();

  select
    p.start_date::date,
    p.vacation_migration_date,
    p.vacation_available_at_migration::numeric
  into
    v_start_date,
    v_migration_date,
    v_initial_available
  from public.profiles p
  where p.id = v_uid;

  if v_start_date is null or p_at is null or p_at < v_start_date then
    return jsonb_build_object(
      'available', 0,
      'granted', 0,
      'used', 0,
      'reserved', 0,
      'next_expiration', null,
      'buckets', jsonb_build_array()
    );
  end if;

  v_has_migration := (v_migration_date is not null);

  -- si hay migration_date, usamos:
  -- - granted = initial_available + (earned_total(p_at) - earned_total(migration_date - 1))
  -- - used/reserved solo desde migration_date
  if v_has_migration then
    v_from_date := greatest(v_start_date, v_migration_date);
  else
    v_from_date := v_start_date;
    v_initial_available := 0;
  end if;

  -- cupo anual (según schema) al momento de evaluación
  annual_days := coalesce(public.vacation_entitlement_days_at_date(v_uid, p_at), 14);

-- ==========================================================
-- 1) EARNED TOTAL @ p_at
-- ==========================================================
v_eval := p_at;

v_cut := (v_start_date + interval '6 months')::date;

-- base correcta: entitlement al momento del primer grant
annual_days :=
  coalesce(public.vacation_entitlement_days_at_date(v_uid, v_cut), 14);

if v_eval < v_cut then
  months_worked :=
    greatest(
      0,
      (extract(year from age(v_eval, v_start_date))::int * 12)
      + extract(month from age(v_eval, v_start_date))::int
    );

  months_worked := least(months_worked, 6);

  earned_at_p := months_worked::numeric;

else

  -- primer grant al cumplir 6 meses
  earned_at_p := annual_days::numeric;

  start_year := extract(year from v_start_date)::int + 1;
  end_year := extract(year from v_eval)::int;

  for y in start_year..end_year loop

    ann := public.anniversary_date(v_start_date, y);

    if ann <= v_eval then
      earned_at_p := earned_at_p
        + coalesce(
            public.vacation_entitlement_days_at_date(v_uid, ann),
            annual_days
          );
    end if;

  end loop;

end if;



-- ==========================================================
-- 2) EARNED TOTAL @ (migration_date - 1)
-- ==========================================================
if v_has_migration then

  v_eval := (v_from_date - interval '1 day')::date;

  if v_eval < v_start_date then

    earned_before_migration := 0;

  else

    v_cut := (v_start_date + interval '6 months')::date;

    annual_days :=
      coalesce(public.vacation_entitlement_days_at_date(v_uid, v_cut), 14);

    if v_eval < v_cut then

      months_worked :=
        greatest(
          0,
          (extract(year from age(v_eval, v_start_date))::int * 12)
          + extract(month from age(v_eval, v_start_date))::int
        );

      months_worked := least(months_worked, 6);

      earned_before_migration := months_worked::numeric;

    else

      earned_before_migration := annual_days::numeric;

      start_year := extract(year from v_start_date)::int + 1;
      end_year := extract(year from v_eval)::int;

      for y in start_year..end_year loop

        ann := public.anniversary_date(v_start_date, y);

        if ann <= v_eval then
          earned_before_migration := earned_before_migration
            + coalesce(
                public.vacation_entitlement_days_at_date(v_uid, ann),
                annual_days
              );
        end if;

      end loop;

    end if;

  end if;

  granted_total :=
    v_initial_available
    + greatest(0, earned_at_p - earned_before_migration);

else

  granted_total := earned_at_p;

end if;

  -- ==========================================================
  -- USED (solo lo ya ocurrido) vs RESERVED (futuro / pendiente)
  -- contando SOLO desde v_from_date (migration_date si existe)
  -- ==========================================================
  select coalesce(sum(
    public.vacation_days_overlap(a.date_from, a.date_to, v_from_date, p_at)
  ), 0)::numeric
  into used_total
  from public.absences a
  where a.user_id = v_uid
    and a.type = 'vacaciones'
    and a.status = 'aprobado'
    and a.date_from <= p_at
    and a.date_to >= v_from_date;

  select coalesce(sum(
    public.vacation_days_overlap(
      a.date_from,
      a.date_to,
      greatest((p_at + interval '1 day')::date, v_from_date),
      a.date_to
    )
  ), 0)::numeric
  into reserved_total
  from public.absences a
  where a.user_id = v_uid
    and a.type = 'vacaciones'
    and a.status in ('pendiente','aprobado')
    and a.date_to > p_at
    and a.date_to >= v_from_date;

  used_total := round(used_total, 2);
  reserved_total := round(reserved_total, 2);

  available_total := greatest(0, round(granted_total - used_total - reserved_total, 2));

  return jsonb_build_object(
    'available', available_total,
    'granted', round(granted_total, 2),
    'used', used_total,
    'reserved', reserved_total,
    'next_expiration', null,
    'buckets', jsonb_build_array()
  );
end;
$$;

create or replace function public.set_absence_status(
  p_absence_id uuid,
  p_status text
)
returns public.absences
language plpgsql
security definer
set search_path = public
as $$
declare
  v_row public.absences;

  v_type text;
  v_from date;
  v_to date;

  v_requested int;
  v_bal jsonb;
  v_available numeric;
begin
  -- ❌ REMOVIDO auth.uid()

  if p_status not in ('pendiente','aprobado','rechazado') then
    raise exception 'Invalid status: %', p_status;
  end if;

  -- ❌ REMOVIDO is_owner()

  -- Traemos datos de la ausencia
  select a.type, a.date_from, a.date_to
    into v_type, v_from, v_to
  from public.absences a
  where a.id = p_absence_id;

  if not found then
    raise exception 'Absence not found';
  end if;

  -- ✅ Validación saldo vacaciones (esto está perfecto)
  if p_status = 'aprobado' and v_type = 'vacaciones' and auth.uid() is not null then
    v_requested := public.vacation_days_between(v_from, v_to);

    v_bal := public.get_my_vacation_balance_at(v_from);
    v_available := coalesce((v_bal->>'available')::numeric, 0);

    if v_requested::numeric > v_available then
      raise exception
        'SALDO_INSUFICIENTE: requeridos=% disponibles=%',
        v_requested,
        v_available;
    end if;
  end if;

  update public.absences a
  set status = p_status
  where a.id = p_absence_id
  returning * into v_row;

  return v_row;
end;
$$;

create or replace function public.get_vacation_balance_for_user_at(
  p_user_id uuid,
  p_at date
)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  v_caller uuid := auth.uid();

  v_start_date date;

  -- MIGRACIÓN
  v_migration_date date;
  v_initial_available numeric := 0;
  v_has_migration boolean := false;

  v_from_date date;

  annual_days int;
  earned_at_p numeric := 0;
  earned_before_migration numeric := 0;
  granted_total numeric := 0;

  used_total numeric := 0;
  reserved_total numeric := 0;
  available_total numeric := 0;

  v_cut date;
  months_worked int;
  y int;
  ann date;
  start_year int;
  end_year int;

  v_eval date;
begin
  if v_caller is null then
    raise exception 'Not authenticated';
  end if;

  if not public.is_owner(v_caller) then
    raise exception 'Not authorized';
  end if;

  if p_user_id is null or p_at is null then
    return jsonb_build_object(
      'available', 0,
      'granted', 0,
      'used', 0,
      'reserved', 0,
      'next_expiration', null,
      'buckets', jsonb_build_array()
    );
  end if;

  select
    p.start_date::date,
    p.vacation_migration_date,
    p.vacation_available_at_migration::numeric
  into
    v_start_date,
    v_migration_date,
    v_initial_available
  from public.profiles p
  where p.id = p_user_id;

  if v_start_date is null or p_at < v_start_date then
    return jsonb_build_object(
      'available', 0,
      'granted', 0,
      'used', 0,
      'reserved', 0,
      'next_expiration', null,
      'buckets', jsonb_build_array()
    );
  end if;

  v_has_migration := (v_migration_date is not null);

  if v_has_migration then
    v_from_date := greatest(v_start_date, v_migration_date);
  else
    v_from_date := v_start_date;
    v_initial_available := 0;
  end if;

  -- ==========================================================
  -- 1) EARNED TOTAL @ p_at  (base correcta: entitlement en v_cut)
  -- ==========================================================
  v_eval := p_at;
  v_cut := (v_start_date + interval '6 months')::date;

  annual_days := coalesce(public.vacation_entitlement_days_at_date(p_user_id, v_cut), 14);

  if v_eval < v_cut then
    months_worked :=
      greatest(
        0,
        (extract(year from age(v_eval, v_start_date))::int * 12)
        + extract(month from age(v_eval, v_start_date))::int
      );
    months_worked := least(months_worked, 6);
    earned_at_p := months_worked::numeric;
  else
    earned_at_p := annual_days::numeric;

    start_year := extract(year from v_start_date)::int + 1;
    end_year := extract(year from v_eval)::int;

    for y in start_year..end_year loop
      ann := public.anniversary_date(v_start_date, y);
      if ann <= v_eval then
        earned_at_p := earned_at_p
          + coalesce(public.vacation_entitlement_days_at_date(p_user_id, ann), annual_days);
      end if;
    end loop;
  end if;

  -- ==========================================================
  -- 2) EARNED TOTAL @ (migration_date - 1)
  -- ==========================================================
  if v_has_migration then
    v_eval := (v_from_date - interval '1 day')::date;

    if v_eval < v_start_date then
      earned_before_migration := 0;
    else
      v_cut := (v_start_date + interval '6 months')::date;
      annual_days := coalesce(public.vacation_entitlement_days_at_date(p_user_id, v_cut), 14);

      if v_eval < v_cut then
        months_worked :=
          greatest(
            0,
            (extract(year from age(v_eval, v_start_date))::int * 12)
            + extract(month from age(v_eval, v_start_date))::int
          );
        months_worked := least(months_worked, 6);
        earned_before_migration := months_worked::numeric;
      else
        earned_before_migration := annual_days::numeric;

        start_year := extract(year from v_start_date)::int + 1;
        end_year := extract(year from v_eval)::int;

        for y in start_year..end_year loop
          ann := public.anniversary_date(v_start_date, y);
          if ann <= v_eval then
            earned_before_migration := earned_before_migration
              + coalesce(public.vacation_entitlement_days_at_date(p_user_id, ann), annual_days);
          end if;
        end loop;
      end if;
    end if;

    granted_total := v_initial_available + greatest(0, earned_at_p - earned_before_migration);
  else
    granted_total := earned_at_p;
  end if;

  -- ==========================================================
  -- USED / RESERVED desde v_from_date
  -- ==========================================================
  select coalesce(sum(
    public.vacation_days_overlap(a.date_from, a.date_to, v_from_date, p_at)
  ), 0)::numeric
  into used_total
  from public.absences a
  where a.user_id = p_user_id
    and a.type = 'vacaciones'
    and a.status = 'aprobado'
    and a.date_from <= p_at
    and a.date_to >= v_from_date;

  select coalesce(sum(
    public.vacation_days_overlap(
      a.date_from,
      a.date_to,
      greatest((p_at + interval '1 day')::date, v_from_date),
      a.date_to
    )
  ), 0)::numeric
  into reserved_total
  from public.absences a
  where a.user_id = p_user_id
    and a.type = 'vacaciones'
    and a.status in ('pendiente','aprobado')
    and a.date_to > p_at
    and a.date_to >= v_from_date;

  used_total := round(used_total, 2);
  reserved_total := round(reserved_total, 2);

  available_total := greatest(0, round(granted_total - used_total - reserved_total, 2));

  return jsonb_build_object(
    'available', available_total,
    'granted', round(granted_total, 2),
    'used', used_total,
    'reserved', reserved_total,
    'next_expiration', null,
    'buckets', jsonb_build_array()
  );
end;
$$;

revoke all on function public.get_vacation_balance_for_user_at(uuid, date) from public;
grant execute on function public.get_vacation_balance_for_user_at(uuid, date) to authenticated;
